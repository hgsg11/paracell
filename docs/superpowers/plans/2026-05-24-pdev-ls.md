# pdev ls Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add `pdev ls`, a read-only CLI command that lists created virtual environments by name and template.

**Architecture:** The command reads existing cells through `CellStatePort` and does not inspect Docker, tmux, or git. A new `ListCellsUseCase` returns `[]domain.Cell`, while the app layer owns command parsing and the output adapter owns table formatting. `Run` keeps the CLI path direct and writes list output to `os.Stdout`.

**Tech Stack:** Go 1.26, standard library, existing Clean Architecture package layout.

---

package output

import (
	"strings"

	"github.com/shige1114/paradev/internal/domain"
)

func FormatCellList(cells []domain.Cell) string {
	var b strings.Builder
	b.WriteString("NAME\tTEMPLATE\n")
	for _, cell := range cells {
		b.WriteString(cell.Name)
		b.WriteByte('\t')
		b.WriteString(cell.Template)
		b.WriteByte('\n')
	}
	return b.String()
}
```

func TestRunはLsでStateのCell一覧を出力する(t *testing.T) {

func TestRunはLsでStateがなくてもヘッダーだけ出力する(t *testing.T) {
	dir := t.TempDir()

	output, err := captureStdout(func() error {
		return Run(context.Background(), []string{"ls"}, dir)
	})

	if err != nil {
		t.Fatalf("Runでエラーが返った: %v", err)
	}
	want := "NAME\tTEMPLATE\n"
	if output != want {
		t.Fatalf("output = %q, want %q", output, want)
	}
	if _, err := os.Stat(filepath.Join(dir, ".pdev", "state.json")); !os.IsNotExist(err) {
		t.Fatalf("state.json existence error = %v, want not exist", err)
	}
}

func captureStdout(fn func() error) (string, error) {
	original := os.Stdout
	read, write, err := os.Pipe()
	if err != nil {
		return "", err
	}
	os.Stdout = write
	defer func() {
		os.Stdout = original
	}()

	runErr := fn()
	closeErr := write.Close()
	output, readErr := io.ReadAll(read)
	if closeErr != nil && runErr == nil {
		runErr = closeErr
	}
	if readErr != nil && runErr == nil {
		runErr = readErr
	}
	return string(output), runErr
}
```
